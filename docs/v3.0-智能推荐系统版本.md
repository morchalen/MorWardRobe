**发布日期**: 2026-Q3

## 一、版本定位与价值跃迁

### 从工具到智能助手的进化
```
v2.0: 数字化工具 → 用户手动决策
         ↓ 智能化升级

```

- **今日穿什么（OOTD）**: 系统根据天气、日程、场合自动生成穿搭方案

- **智能搭配建议**: 基于色彩理论+时尚规则的组合推荐

### 用户价值提升
✅ **决策减负**: 每天30秒获取专业级穿搭建议
✅ **所见即所得**: 试穿确认后再决定，减少试错成本

```python
# services/recommendation/collaborative.py
class ImplicitCollaborativeFiltering:
    """
    基于隐式反馈的协同过滤
    反馈信号：
    - 点击查看详情 (weight=1)
    - 加入收藏 (weight=3)
    - 实际穿着 (weight=5)
    - 虚拟试穿 (weight=4)
    """

    def build_interaction_matrix(self, user_id: str):
        """构建用户-物品交互矩阵"""
        interactions = db.query("""
            SELECT clothing_id,
                   SUM(CASE
                       WHEN action = 'view' THEN 1
                       WHEN action = 'favorite' THEN 3
                       WHEN action = 'wear' THEN 5
                       WHEN action = 'try_on' THEN 4
                   END) as score
            FROM user_actions
            WHERE user_id = $1
              AND created_at > NOW() - INTERVAL '90 days'
            GROUP BY clothing_id
        """, user_id)

        return scipy.sparse.csr_matrix(interactions)

    def recommend_similar_users(self, target_user, k=20):
        """用户-Based CF：找到相似用户喜欢的物品"""
        similar_users = cosine_similarity(
            target_user.interaction_matrix,
            all_users.interaction_matrix
        ).argsort()[-k:]

        recommendations = []
        for sim_user in similar_users:
            items = sim_user.get_highly_rated_items()
            items_not_owned_by_target = [
                item for item in items
                if item not in target_user.owned_items
            ]
            recommendations.extend(items_not_owned_by_target)

        return self.rerank_by_diversity(recommendations)
```

```typescript
// services/recommendation/context.ts
interface ContextSignal {
  weather: {
    temperature: number;       // 当前温度 °C
    condition: 'sunny' | 'cloudy' | 'rainy' | 'snowy';
    humidity: number;          // 湿度 %
    wind_speed: number;        // 风速 km/h
  };
  occasion: {
    type: 'work' | 'casual' | 'formal' | 'sports' | 'date' | 'travel';
    indoor_outdoor: 'indoor' | 'outdoor' | 'mixed';
    formality_level: number;   // 1-10, 1最随意
  };
  time_context: {
    hour: number;              // 24小时制
    day_of_week: 'weekday' | 'weekend';
    season: 'spring' | 'summer' | 'autumn' | 'winter';
  };
  user_state: {
    mood?: string;             // 用户自选心情标签
    energy_level?: 'low' | 'medium' | 'high'; // 用户选择
    special_constraints?: string[]; // ["need_waterproof", "no_white_after_labor_day"]
  };
}

class ContextualRecommender {
  async generateOutfit(context: ContextSignal): Promise<OutfitRecommendation[]> {
    // Step 1: 根据天气筛选合适季节的衣物
    const weatherFiltered = await this.filterByWeather(context.weather);

    // Step 2: 根据场合筛选风格兼容性
    const occasionFiltered = await this.filterByOccasion(context.occasion);

    // Step 3: 结合用户身材数据做版型适配评分
    const fitScored = await this.scoreByBodyFit(
      intersection(weatherFiltered, occasionFiltered),
      await this.getUserProfile()
    );

    // Step 4: 应用时尚规则进行配色和谐度打分
    const colorHarmonized = await this.applyColorRules(fitScored);

    // Step 5: 生成多套候选方案（Top-K多样化）
    const candidates = this.generateDiverseOutfits(colorHarmonized, k=5);

    // Step 6: 对每套方案添加解释说明
    return candidates.map(outfit => ({
      ...outfit,
      explanation: this.generateExplanation(outfit, context),
      confidence_score: this.calculateConfidence(outfit)
    }));
  }
}
```

#### IDM-VTON集成流程

```python
# services/virtual_tryon/pipeline.py
class VirtualTryOnPipeline:
    def __init__(self):
        self.vton_model = load_idm_vton_model()
        self.pose_estimator = load_pose_estimator('HRNet-W48')

    async def try_on_clothing(
        self,
        person_image: Image,           # 用户全身照
        garment_image: Image,          # 衣橱中的抠图衣物
        category: str,                 # upper_body/lower_body/dresses
        pose_control: PoseControl = None,  # 可选姿态调整
        preserve_face: bool = True     # 保护面部特征不被修改
    ) -> TryOnResult:

        # Step 1: 人体关键点检测
        pose_keypoints = await self.pose_estimator.detect(person_image)

        # Step 2: 衣物形变预处理（根据人体尺寸自适应）
        warped_garment = self.warp_garment_to_body(
            garment_image,
            pose_keypoints,
            body_measurements=await self.get_user_body_data()
        )

        # Step 3: IDM-VTON推理生成换装结果
        try_on_result = await self.vton_model.inference(
            person_image=person_image,
            garment_image=warped_garment,
            category=category,
            pose_keypoints=pose_keypoints
        )

        # Step 4: 后处理（面部保护、边缘融合）
        if preserve_face:
            try_on_result = self.restore_original_face(
                try_on_result,
                original_image=person_image,
                face_mask=self.extract_face_mask(person_image)
        )

        # Step 5: 质量评估与增强
        quality_score = self.assess_generation_quality(try_on_result)
        if quality_score < 0.7:
            try_on_result = self.enhance_result(try_on_result)

        return TryOnResult(
            image=try_on_result,
            quality_score=quality_score,
            processing_time_ms=self.timer.elapsed(),
            garment_fit_analysis=self.analyze_fit(pose_keypoints, category)
        )
```

### 3.2 🔍 以图搜衣与相似款发现（P1优先级）

#### 技术实现：CLIP视觉嵌入

```python
# services/image_search/clip_searcher.py
class VisualSearchEngine:
    def __init__(self):
        self.clip_model = CLIPModel.from_pretrained("openai/clip-vit-large-patch14")
        self.index = FaissIndexHNSWFlat(d=768, M=32)  # 高维向量索引

    async def index_clothing(self, clothing_id: str, image: Image):
        """为每件衣物生成并存储视觉特征向量"""
        with torch.no_grad():
            image_input = self.clip_processor(images=image, return_tensors="pt")
            embedding = self.clip_model.get_image_features(**image_input)
            embedding = F.normalize(embedding, p=2, dim=1)

        self.index.add(embedding.numpy())
        redis.set(f"embed:{clothing_id}", embedding.numpy().tobytes())

    async def search_by_image(self, query_image: Image, top_k: int = 12) -> SearchResults:
        """以图搜图：找到视觉相似的衣物"""
        query_embedding = self._encode_image(query_image)
        distances, indices = self.index.search(query_embedding, top_k)

        results = []
        for dist, idx in zip(distances[0], indices[0]):
            clothing_id = self.id_map[idx]
            similarity = 1 / (1 + dist)  # 转换为相似度分数

            results.append({
                'clothing_id': clothing_id,
                'similarity_score': round(similarity, 4),
                'matched_attributes': self._explain_similarity(query_embedding, idx)
            })

        return sorted(results, key=lambda x: x['similarity_score'], reverse=True)
```

```sql
    user_rating INTEGER CHECK (user_rating BETWEEN 1 AND 5), -- 用户事后评分
    user_feedback TEXT,               -- 文字反馈："太热了"、"被夸好看"
    photo_evidence TEXT,              -- 穿搭实拍图（可选上传）
    created_at TIMESTAMPTZ DEFAULT NOW()
);
CREATE INDEX idx_wear_logs_user_date ON outfit_wear_logs(user_id, wear_date DESC);

-- 衣物视觉向量索引元数据表（配合Faiss/Milvus）
CREATE TABLE clothing_embeddings (
    clothing_id UUID PRIMARY KEY REFERENCES clothes(id) ON DELETE CASCADE,
    model_version VARCHAR(20) NOT NULL,
    embedding_vector BYTEA NOT NULL,  -- 二进制存储768维向量
    embedding_updated_at TIMESTAMPTZ DEFAULT NOW()
);

-- 用户隐式反馈表（用于协同过滤）
CREATE TABLE implicit_feedback (
    id BIGSERIAL PRIMARY KEY,
    user_id UUID REFERENCES users(id),
    clothing_id UUID REFERENCES clothes(id),
    action_type VARCHAR(30) NOT NULL, -- view/favorite/try_on/wear/search_click
    action_value FLOAT NOT NULL,      -- 权重值
    context JSONB,                    -- 发生时的上下文（页面来源等）
    created_at TIMESTAMPTZ DEFAULT NOW()
);
CREATE INDEX idx_implicit_fb_user_clothing ON implicit_feedback(user_id, clothing_id);
CREATE INDEX idx_implicit_fb_user_time ON implicit_feedback(user_id, created_at DESC);
-- 分区表：按月分区，提升查询性能
```

## 六、性能优化策略

### 6.1 推荐系统性能

| 场景 | 目标延迟 | 优化手段 |
|------|---------|---------|
| OOTD首页加载 | < 800ms | 预计算缓存 + 异步刷新 |
| 搭配评分实时计算 | < 500ms | 规则引擎轻量化 + 结果缓存 |
| 以图搜图 | < 1s | Faiss HNSW索引 + GPU加速 |
| 虚拟试衣生成 | < 10s | 模型量化 + 批量推理 |

### 6.2 向量检索加速

```python
# 使用Milvus或Qdrant管理十亿级向量
from pymilvus import Collection, FieldSchema, CollectionSchema

fields = [
    FieldSchema(name="id", dtype=DataType.INT64, is_primary=True),
    FieldSchema(name="clothing_id", dtype=DataType.VARCHAR, max_length=36),
    FieldSchema(name="embedding", dtype=DataType.FLOAT_VECTOR, dim=768)
]

schema = CollectionSchema(fields, "clothing visual embeddings")
collection = Collection(name="clothing_embeddings", schema=schema)

# 创建HNSW索引（高召回率+低延迟）
index_params = {
    "metric_type": "COSINE",
    "index_type": "HNSW",
    "params": {"M": 32, "efConstruction": 256}
}
collection.create_index(field_name="embedding", index_params=index_params)
```

## 八、已知限制与技术债务

1. **冷启动问题**: 新用户无历史数据时推荐精度依赖规则引擎（需引导期）
2. **虚拟试衣局限**: 复杂姿势/极端体型准确率下降（需更多训练数据）
3. **算力成本高**: VTON模型单次推理GPU消耗大（需考虑成本控制策略）
4. **时尚趋势滞后**: 规则引擎需人工定期更新（考虑引入LLM解析时尚资讯）
5. **隐私权衡**: 上下文感知需位置/日历权限（需明确告知用户）
