```bash
echo "=== SmartWardrobe 开发环境前置检查 ==="

    if command -v $1 &> /dev/null; then
        version=$($1 --version 2>&1 | head -n1)
        echo "✅ $1: $version"

        echo "❌ $1: 未安装"
```

## 四、代码规范与最佳实践

### 4.1 Go后端规范

```go
// 项目标准Handler模式
package handler

import (
    "net/http"

    "github.com/gin-gonic/gin"
    "smartwardrobe/internal/model"
    "smartwardrobe/internal/service"
    "smartwardrobe/pkg/response"
)

type ClothingHandler struct {
    clothingService service.ClothingServiceInterface
}

func NewClothingHandler(svc service.ClothingServiceInterface) *ClothingHandler {
    return &ClothingHandler{clothingService: svc}
}

// @Summary 上传衣物
// @Description 支持单张或批量上传衣物图片及元数据
// @Tags 衣物管理
// @Accept multipart/form-data
// @Produce json
// @Param files formData file true "图片文件"
// @Param category formData string true "品类"
// @Success 201 {object} response.Response{data=model.UploadResult}
// @Failure 400 {object} response.ErrorResponse
// @Router /api/v1/clothes/upload [post]
func (h *ClothingHandler) Upload(c *gin.Context) {
    ctx := c.Request.Context()
    userID := middleware.GetUserID(ctx)  // 从中间件获取当前用户ID

    // 1. 参数校验（使用validator库）
    var req model.UploadRequest
    if err := c.ShouldBind(&req); err != nil {
        response.ValidationError(c, err)
        return
    }

    // 2. 业务逻辑委托给Service层
    result, err := h.clothingService.UploadClothing(ctx, userID, req)
    if err != nil {
        // 错误分类处理
        switch {
        case errors.Is(err, service.ErrWardrobeFull):
            response.Error(c, http.StatusConflict, "WARDROBE_FULL", "衣橱容量已达上限")
        case errors.Is(err, service.ErrInvalidFile):
            response.Error(c, http.StatusBadRequest, "INVALID_FILE", err.Error())
        default:
            response.InternalError(c, err)
        }
        return
    }

    // 3. 成功响应
    response.Success(c, http.StatusCreated, result)
}

// 分页查询标准模式
func (h *ClothingHandler) List(c *gin.Context) {
    ctx := c.Request.Context()

    var query model.ClothingListQuery
    if err := c.ShouldBindQuery(&query); err != nil {
        response.ValidationError(c, err)
```

```go
package service

import (
    "context"
    "time"

    "smartwardrobe/internal/repository"
    "smartwardrobe/pkg/storage"
)

type ClothingServiceImpl struct {
    repo   repository.ClothingRepository
    storage storage.ObjectStorage
}

func (s *ClothingServiceImpl) UploadClothing(
    ctx context.Context,
    userID string,
    req model.UploadRequest,
) (*model.UploadResult, error) {
    // 1. 业务规则校验
    count, err := s.repo.CountActiveByUser(ctx, userID)
    if err != nil {
        return nil, fmt.Errorf("count clothes: %w", err)
    }

    limit := s.getUserWardrobeLimit(userID)
    if count+len(req.Files) > limit {
        return nil, ErrWardrobeFull
    }

    // 2. 处理图片上传（事务内）
    var clothingItems []*model.Clothing

    err = s.repo.Transaction(ctx, func(txRepo repository.Repository) error {
        for _, file := range req.Files {
            // 上传到对象存储
            uploadResult, err := s.storage.Upload(ctx, file, storage.UploadOptions{
                Category: req.Category,
                UserID:   userID,
            })
            if err != nil {
                return fmt.Errorf("upload image %s: %w", file.Filename, err)
            }

            // 创建数据库记录
            clothing := &model.Clothing{
                UserID:        userID,
                Name:          req.Name,
                Category:      req.Category,
                PrimaryColor:  req.PrimaryColor,
                ImageURL:      uploadResult.URL,
                ThumbnailURL:  uploadResult.ThumbnailURL,
                ImageHash:     uploadResult.Hash,
                Status:        "active",
            }

            if err := txRepo.Create(ctx, clothing); err != nil {
                return fmt.Errorf("create clothing record: %w", err)
            }

            clothingItems = append(clothingItems, clothing)
        }
        return nil
    })

    if err != nil {
        return nil, err
    }

    // 3. 异步触发AI处理（如果用户有权限）
    if s.userHasAIAccess(userID) {
        clothingIDs := extractIDs(clothingItems)
        go s.triggerBatchAIProcessing(ctx, userID, clothingIDs)
    }

    return &model.UploadResult{
        ClothingIDs:   extractIDs(clothingItems),
        ProcessedImages: mapToImageResults(clothingItems),
        Status:        "success",
    }, nil
}
```

## 九、安全合规要求

### 9.1 必须遵守的安全 Checklist

- [ ] **无明文密码**: 所有密码bcrypt hash存储（cost≥12）
- [ ] **SQL注入防护**: 全部参数化查询，禁止字符串拼接SQL
- [ ] **XSS防护**: 输出转义 + CSP策略
- [ ] **CSRF保护**: Stateful Token验证
- [ ] **认证Token安全**: JWT使用RS256签名，Access Token有效期≤15分钟
- [ ] **HTTPS强制**: 生产环境TLS 1.3 only
- [ ] **文件上传安全**: 类型白名单 + 大小限制 + 病毒扫描
- [ ] **敏感数据脱敏**: 日志中不记录密码/token/PII
- [ ] **权限最小化**: 数据库账号只授予必要权限
- [ ] **依赖安全**: 定期运行`npm audit` / `go vuln` / `pip-audit`
- [ ] **CORS配置**: 仅允许信任域名
- [ ] **Rate Limiting**: 所公开API实施限流
- [ ] **隐私合规**: GDPR/PIPL要求的用户数据导出/删除接口

### 9.2 敏感数据处理规范

```typescript
// ❌ 错误：日志中泄露敏感信息
logger.info(`User ${user.email} logged in from IP ${ip}`);

// ✅ 正确：脱敏处理
logger.info(`User ${maskEmail(user.email)} logged in`);

function maskEmail(email: string): string {
  const [local, domain] = email.split('@');
  return `${local[0]}***@${domain}`;
}
```

## 十一、问题排查指南

### 11.1 常见问题FAQ

**Q: 启动时报 "port 8080 already in use"**
```bash
# 查找占用端口的进程
lsof -i :8080
# 或 Windows
netstat -ano | findstr :8080

# 杀掉进程（谨慎操作）
kill -9 <PID>
```

**Q: 前端请求后端API报 CORS错误**
```bash
# 确认Vite代理配置正确
# vite.config.ts 中应有:
server: {
  proxy: {
    '/api': {
      target: 'http://localhost:8080',
      changeOrigin: true,
    },
  },
}
```

**Q: AI服务加载模型很慢**
```bash
# 检查是否有GPU可用
nvidia-smi

# 如果没有GPU，修改配置使用CPU模式
# .env 中设置: DEVICE=cpu
# 首次下载模型会较慢，后续有缓存
```

**Q: 数据库迁移失败**
```bash
# 查看当前迁移版本
cd apps/api-gateway && go run cmd/migrate/main.go version

# 回滚到上一个版本
go run cmd/migrate/main.go down 1

# 强制重新应用（慎用！会丢失数据）
go run cmd/migrate/main.go force version <target_version>
```
