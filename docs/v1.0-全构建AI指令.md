# SmartWardrobe v1.0 全构建 AI 实施指令

## 📋 文档说明
**目标**: 让 AI 助手（如 Claude/GPT/DeepSeek）能够**零人工干预**完成 v1.0 版本的全部代码编写与项目搭建
**适用阶段**: 仅前后端基础功能（AI 模块延后接入）
**技术栈确认**:
- 前端: React 18 + TypeScript + Vite + Tailwind CSS + Zustand + React Router + Axios
- 后端: Go 1.21+ + Gin Framework (端口 8080)
- 数据库: PostgreSQL (开发可用 SQLite)
- 存储: MinIO / 本地文件系统
- 端口分配: 前端3000 / 后端8080 / AI预留8000

#### 5.2 创建API服务层 `src/services/api.ts`

```typescript
import axios from 'axios';
import type { ApiResponse, PaginatedResponse, ApiError } from '@/types';

const apiClient = axios.create({
  baseURL: '/api/v1',  // 通过Vite代理到后端8080
  timeout: 15000,
  headers: {
    'Content-Type': 'application/json',
  },
});

// 请求拦截器：附加Token
apiClient.interceptors.request.use((config) => {
  const token = localStorage.getItem('access_token');
  if (token) {
    config.headers.Authorization = `Bearer ${token}`;
  }
  config.headers['X-Request-ID'] = crypto.randomUUID();
  return config;
});

// 响应拦截器：统一错误处理
apiClient.interceptors.response.use(
  (response) => {
    const data = response.data as ApiResponse;
    if (data.code >= 40000) {
      return Promise.reject(new Error(data.message));
    }
    return response;
  },
  (error) => {
    if (error.response) {
      const data = error.response.data as ApiError;
      
      // Token过期处理
      if (error.response.status === 401 && data?.error === 'AUTH_TOKEN_EXPIRED') {
        return refreshTokenAndRetry(error.config);
      }
      
      return Promise.reject(new Error(data?.message || '请求失败'));
    }
    return Promise.reject(new Error('网络异常'));
  }
);

async function refreshTokenAndRetry(originalRequest: any) {
  const refreshToken = localStorage.getItem('refresh_token');
  try {
    const response = await axios.post('/api/v1/auth/refresh', {
      refresh_token: refreshToken,
    });
    const { access_token, refresh_token: newRefresh } = response.data.data;
    
    localStorage.setItem('access_token', access_token);
    localStorage.setItem('refresh_token', newRefresh);
    
    originalRequest.headers.Authorization = `Bearer ${access_token}`;
    return apiClient(originalRequest);
  } catch (err) {
    // Refresh失败，跳转登录
    localStorage.removeItem('access_token');
    localStorage.removeItem('refresh_token');
    window.location.href = '/login';
    return Promise.reject(err);
  }
}

// ============================================================
// 认证API
// ============================================================
export const authApi = {
  register: (data: { email: string; password: string; nickname: string }) =>

  update: (id: string, data: Partial<Clothing>) =>
```

#### 5.3 创建Zustand状态管理 `src/stores/index.ts`

```typescript
import { create } from 'zustand';
import { persist } from 'zustand/middleware';
import type { User, Clothing, Folder } from '@/types';
import { authApi, clothesApi, foldersApi } from '@/services/api';

// ============================================================
// 认证状态
// ============================================================
interface AuthState {
  user: User | null;
  accessToken: string | null;
  isAuthenticated: boolean;
  isLoading: boolean;

  login: (email: string, password: string) => Promise<void>;
  register: (email: string, password: string, nickname: string) => Promise<void>;
  logout: () => Promise<void>;
  fetchUser: () => Promise<void>;
}

export const useAuthStore = create<AuthState>()(
  persist(
    (set, get) => ({
      user: null,
      accessToken: null,
      isAuthenticated: false,
      isLoading: false,

      login: async (email, password) => {
        set({ isLoading: true });
        try {
          const res = await authApi.login({ email, password });
          const { access_token, refresh_token, user } = res.data.data;
          
          localStorage.setItem('access_token', access_token);
          localStorage.setItem('refresh_token', refresh_token);
          
          set({ user, accessToken: access_token, isAuthenticated: true, isLoading: false });
        } catch (err: any) {
          set({ isLoading: false });
          throw err;
        }
      },

      register: async (email, password, nickname) => {
        set({ isLoading: true });
        try {
          await authApi.register({ email, password, nickname });
          // 注册成功后自动登录
          get().login(email, password);
        } catch (err: any) {
          set({ isLoading: false });
          throw err;
        }
      },

      logout: async () => {
        try {
          await authApi.logout();
        } finally {
          localStorage.removeItem('access_token');
          localStorage.removeItem('refresh_token');
          set({ user: null, accessToken: null, isAuthenticated: false });
        }
      },

      fetchUser: async () => {
        const token = localStorage.getItem('access_token');
        if (!token) return;
        
        try {
          const res = await authApi.getMe();
          set({ user: res.data.data, isAuthenticated: true, accessToken: token });
        } catch {
          // Token无效，清除状态
          localStorage.removeItem('access_token');
          set({ user: null, accessToken: null, isAuthenticated: false });
        }
      },
    }),
    {
      name: 'auth-storage',
      partialize: (state) => ({ user: state.user, isAuthenticated: state.isAuthenticated }),
    }
  )
);

// ============================================================
// 衣物状态
// ============================================================
interface WardrobeState {
  clothes: Clothing[];
  folders: Folder[];
  selectedClothing: Clothing | null;
  isLoading: boolean;
  currentPage: number;
  totalItems: number;
  filters: {
    category: string | null;
    folderId: string | null;
    searchQuery: string;
    showFavoritesOnly: boolean;
  };

  fetchClothes: (page?: number) => Promise<void>;
  fetchFolders: () => Promise<void>;
  setSelectedClothing: (clothing: Clothing | null) => void;
  toggleFavorite: (id: string) => Promise<void>;
  deleteClothing: (id: string) => Promise<void>;
  setFilter: (key: string, value: any) => void;
  clearFilters: () => void;
}

export const useWardrobeStore = create<WardrobeState>()((set, get) => ({
  clothes: [],
  folders: [],
  selectedClothing: null,
  isLoading: false,
  currentPage: 1,
  totalItems: 0,
  filters: {
    category: null,
    folderId: null,
    searchQuery: '',
    showFavoritesOnly: false,
  },

  fetchClothes: async (page = 1) => {
    set({ isLoading: true });
    try {
      const { filters } = get();
      const params: any = { page, per_page: 20 };
      if (filters.category) params.category = filters.category;
      if (filters.folderId) params.folder_id = filters.folderId;
      if (filters.searchQuery) params.search = filters.searchQuery;
      if (filters.showFavoritesOnly) params.is_favorite = true;

      const res = await clothesApi.list(params);
      const { items, pagination } = res.data.data;
      
      set({
        clothes: items,
        currentPage: pagination.page,
        totalItems: pagination.total_items,
        isLoading: false,
      });
    } catch (err) {
      set({ isLoading: false });
      console.error('Failed to fetch clothes:', err);
    }
  },

  fetchFolders: async () => {
    try {
      const res = await foldersApi.list();
      set({ folders: res.data.data });
    } catch (err) {
      console.error('Failed to fetch folders:', err);
    }
  },

  setSelectedClothing: (clothing) => set({ selectedClothing: clothing }),

  toggleFavorite: async (id) => {
    try {
      const res = await clothesApi.toggleFavorite(id);
      const { is_favorite } = res.data.data;
      
      set((state) => ({
        clothes: state.clothes.map(c => 
          c.id === id ? { ...c, is_favorite } : c
        ),
        selectedClothing: state.selectedClothing?.id === id 
          ? { ...state.selectedClothing, is_favorite }
          : state.selectedClothing,
      }));
    } catch (err) {
      console.error('Failed to toggle favorite:', err);
    }
  },

  deleteClothing: async (id) => {
    try {
      await clothesApi.delete(id);
      set((state) => ({
        clothes: state.clothes.filter(c => c.id !== id),
        totalItems: state.totalItems - 1,
      }));
    } catch (err) {
      console.error('Failed to delete clothing:', err);
    }
  },

  setFilter: (key, value) => {
    set((state) => ({
      filters: { ...state.filters, [key]: value },
    }));
    // 自动刷新列表
    get().fetchClothes(1);
  },

  clearFilters: () => {
    set({
      filters: {
        category: null,
        folderId: null,
        searchQuery: '',
        showFavoritesOnly: false,
      },
    });
    get().fetchClothes(1);
  },
}));
```

```typescript
import { clsx, type ClassValue } from 'clsx';
import { twMerge } from 'tailwind-merge';

/**
 * 合并Tailwind类名（解决冲突问题）
 */
export function cn(...inputs: ClassValue[]) {
  return twMerge(clsx(inputs));
}

/**
 * 格式化日期显示
 */
export function formatDate(dateString: string): string {
  const date = new Date(dateString);
  const now = new Date();
  const diffTime = now.getTime() - date.getTime();
  const diffDays = Math.floor(diffTime / (1000 * 60 * 60 * 24));

  if (diffDays === 0) return '今天';
  if (diffDays === 1) return '昨天';
  if (diffDays < 7) return `${diffDays}天前`;
  if (diffDays < 30) return `${Math.floor(diffDays / 7)}周前`;
  if (diffDays < 365) return `${Math.floor(diffDays / 30)}个月前`;
  return date.toLocaleDateString('zh-CN');
}

/**
 * 格式化价格
 */
export function formatPrice(price: number | undefined): string {
  if (!price || price === 0) return '-';
  return `¥${price.toLocaleString('zh-CN', { minimumFractionDigits: 0, maximumFractionDigits: 2 })}`;
}
```

**添加衣物页面** `src/pages/clothes/AddClothingPage.tsx` (简化上传表单):
```tsx
import { useState } from 'react';
import { useNavigate } from 'react-router-dom';
import { Upload, X } from 'lucide-react';
import { clothesApi } from '@/services/api';
import { useWardrobeStore } from '@/stores';
import { CATEGORY_LABELS } from '@/utils';
import type { CategoryType, SeasonType } from '@/types';

export function AddClothingPage() {
  const navigate = useNavigate();
  const { fetchClothes } = useWardrobeStore();

  const [files, setFiles] = useState<File[]>([]);
  const [previews, setPreviews] = useState<string[]>([]);
  const [formData, setFormData] = useState({
    name: '',
    category: 'tops' as CategoryType,
    primary_color: '#000000',
    brand: '',
    price: '',
    seasons: [] as SeasonType[],
    material: '',
    custom_tags: '',
    notes: '',
  });
  const [submitting, setSubmitting] = useState(false);
  const [error, setError] = useState('');

  const handleFileChange = (e: React.ChangeEvent<HTMLInputElement>) => {
    if (e.target.files) {
      const newFiles = Array.from(e.target.files);
      setFiles(prev => [...prev, ...newFiles].slice(0, 10)); // 最多10张
      
      // 生成预览URL
      newFiles.forEach(file => {
        const reader = new FileReader();
        reader.onload = (ev) => {
          setPreviews(prev => [...prev, ev.target.result as string]);
        };
        reader.readAsDataURL(file);
      });
    }
  };

  const removeFile = (index: number) => {
    setFiles(prev => prev.filter((_, i) => i !== index));
    setPreviews(prev => prev.filter((_, i) => i !== index));
  };

  const handleSeasonToggle = (season: SeasonType) => {
    setFormData(prev => ({
      ...prev,
      seasons: prev.seasons.includes(season)
        ? prev.seasons.filter(s => s !== season)
        : [...prev.seasons, season]
    }));
  };

  const handleSubmit = async (e: React.FormEvent) => {
    e.preventDefault();
    setError('');

    if (files.length === 0) {
      setError('请至少选择一张图片');
      return;
    }

    setSubmitting(true);
    try {
      const uploadData = new FormData();
      files.forEach(file => {
        uploadData.append('images', file);
      });
      uploadData.append('category', formData.category);
      uploadData.append('primary_color', formData.primary_color);
      if (formData.name) uploadData.append('name', formData.name);
      if (formData.brand) uploadData.append('brand', formData.brand);
      if (formData.price) uploadData.append('price', formData.price);
      if (formData.seasons.length > 0) {
        formData.seasons.forEach(s => uploadData.append('seasons[]', s));
      }
      if (formData.material) uploadData.append('material', formData.material);
      if (formData.custom_tags) {
        formData.custom_tags.split(',').forEach(t => {
          if (t.trim()) uploadData.append('custom_tags[]', t.trim());
        });
      }

      await clothesApi.create(uploadData);
      fetchClothes(); // 刷新列表
      navigate('/', { replace: true });
    } catch (err: any) {
      setError(err.message || '上传失败，请重试');
    } finally {
      setSubmitting(false);
    }
  };

  return (
    <div className="max-w-2xl mx-auto space-y-6">
      <h1 className="text-2xl font-bold text-neutral-900">添加新衣物</h1>

      <form onSubmit={handleSubmit} className="space-y-6">
        {error && (
          <div className="p-3 rounded-lg bg-red-50 border border-red-200 text-red-700 text-sm">
            {error}
          </div>
        )}

        {/* 图片上传区域 */}
        <div>
          <label className="block text-sm font-medium text-neutral-700 mb-2">
            衣物图片 <span className="text-red-500">*</span> (最多10张)
          </label>
          <div className="border-2 border-dashed border-neutral-300 rounded-xl p-6 text-center hover:border-primary-400 transition-colors">
            <input
              type="file"
              multiple
              accept="image/*"
              onChange={handleFileChange}
              className="hidden"
              id="file-upload"
            />
            <label htmlFor="file-upload" className="cursor-pointer">
              <Upload className="w-10 h-10 text-neutral-400 mx-auto mb-2" />
              <p className="text-sm text-neutral-600">点击或拖拽图片到此区域</p>
              <p className="text-xs text-neutral-400 mt-1">支持 JPG, PNG, WebP，单张最大10MB</p>
            </label>

            {/* 预览已选择的图片 */}
            {previews.length > 0 && (
              <div className="grid grid-cols-5 gap-3 mt-4">
                {previews.map((preview, idx) => (
                  <div key={idx} className="relative group aspect-square rounded-lg overflow-hidden bg-neutral-100">
                    <img src={preview} alt="" className="w-full h-full object-cover" />
                    <button
                      type="button"
                      onClick={() => removeFile(idx)}
                      className="absolute top-1 right-1 w-6 h-6 bg-red-500 text-white rounded-full opacity-0 group-hover:opacity-100 transition-opacity flex items-center justify-center"
                    >
                      <X className="w-3.5 h-3.5" />
                    </button>
                  </div>
                ))}
              </div>
            )}
          </div>
        </div>

        {/* 基本信息 */}
        <div className="grid grid-cols-2 gap-4">
          <div>
            <label className="block text-sm font-medium text-neutral-700 mb-1.5">名称 (可选)</label>
            <input
              type="text"
              value={formData.name}
              onChange={e => setFormData(prev => ({ ...prev, name: e.target.value }))}
              placeholder="例如: ZARA黑色羊毛大衣"
              className="input-base"
            />
          </div>
          <div>
            <label className="block text-sm font-medium text-neutral-700 mb-1.5">品类 *</label>
            <select
              value={formData.category}
              onChange={e => setFormData(prev => ({ ...prev, category: e.target.value as CategoryType }))}
              className="input-base"
            >
              {Object.entries(CATEGORY_LABELS).map(([val, label]) => (
                <option key={val} value={val}>{label}</option>
              ))}
            </select>
          </div>
        </div>

        <div className="grid grid-cols-3 gap-4">
          <div>
            <label className="block text-sm font-medium text-neutral-700 mb-1.5">品牌 (可选)</label>
            <input
              type="text"
              value={formData.brand}
              onChange={e => setFormData(prev => ({ ...prev, brand: e.target.value }))}
              placeholder="例如: ZARA, UNIQLO"
              className="input-base"
            />
          </div>
          <div>
            <label className="block text-sm font-medium text-neutral-700 mb-1.5">价格 (可选)</label>
            <input
              type="number"
              value={formData.price}
              onChange={e => setFormData(prev => ({ ...prev, price: e.target.value }))}
              placeholder="0.00"
              step="0.01"
              className="input-base"
            />
          </div>
          <div>
            <label className="block text-sm font-medium text-neutral-700 mb-1.5">主色 *</label>
            <div className="flex items-center gap-2">
              <input
                type="color"
                value={formData.primary_color}
                onChange={e => setFormData(prev => ({ ...prev, primary_color: e.target.value }))}
                className="w-10 h-10 rounded cursor-pointer border border-neutral-200"
              />
              <input
                type="text"
                value={formData.primary_color}
                onChange={e => setFormData(prev => ({ ...prev, primary_color: e.target.value }))}
                className="input-base flex-1"
                placeholder="#000000"
              />
            </div>
          </div>
        </div>

        {/* 季节选择 */}
        <div>
          <label className="block text-sm font-medium text-neutral-700 mb-2">适用季节</label>
          <div className="flex gap-2">
            {(['spring', 'summer', 'autumn', 'winter'] as SeasonType[]).map(season => (
              <button
                key={season}
                type="button"
                onClick={() => handleSeasonToggle(season)}
                className={`px-4 py-2 rounded-lg text-sm font-medium transition-colors ${
                  formData.seasons.includes(season)
                    ? 'bg-primary-100 text-primary-700 border-primary-200'
                    : 'bg-neutral-100 text-neutral-600 hover:bg-neutral-200'
                }`}
              >
                {{ spring: '春', summer: '夏', autumn: '秋', winter: '冬' }[season]}
              </button>
            ))}
          </div>
        </div>

        {/* 材质和标签 */}
        <div className="grid grid-cols-2 gap-4">
          <div>
            <label className="block text-sm font-medium text-neutral-700 mb-1.5">材质 (可选)</label>
            <input
              type="text"
              value={formData.material}

            />

        {/* 提交按钮 */}
        <div className="flex gap-3 pt-4">

}
```

```go
package response

import (
	"net/http"

	"github.com/gin-gonic/gin"
)

type Response struct {
	Code    int         `json:"code"`
	Message string      `json:"message"`
	Data    interface{} `json:"data,omitempty"`
	Meta    *Meta       `json:"meta,omitempty"`
}

type Meta struct {
	RequestID string `json:"request_id"`
	Timestamp string `json:"timestamp"`
}

type PaginatedData struct {
	Items      interface{} `json:"items"`
	Pagination Pagination  `json:"pagination"`
}

type Pagination struct {
	Page       int `json:"page"`
	PerPage    int `json:"per_page"`
	TotalItems int `json:"total_items"`
	TotalPages int `json:"total_pages"`
	HasNext    bool `json:"has_next"`
	HasPrev    bool `json:"has_prev"`
}

func Success(c *gin.HContext, data interface{}) {
	c.JSON(http.StatusOK, Response{
		Code:    200,
		Message: "success",
		Data:    data,
		Meta: &Meta{
			RequestID: c.GetHeader("X-Request-ID"),
			Timestamp: currentTime(),
		},
	})
}

func Created(c *gin.HContext, data interface{}) {
	c.JSON(http.StatusCreated, Response{
		Code:    201,
		Message: "created",
		Data:    data,
		Meta: &Meta{
			RequestID: c.GetHeader("X-Request-ID"),
			Timestamp: currentTime(),
		},
	})
}

func NoContent(c *gin.HContext) {
	c.Status(http.StatusNoContent)
}

func Paginated(c *gin.HContext, items interface{}, page, perPage, total int) {
	totalPages := total / perPage
	if total%perPage > 0 {
		totalPages++
	}

	c.JSON(http.StatusOK, Response{
		Code:    200,
		Message: "success",
		Data: PaginatedData{
			Items: items,
			Pagination: Pagination{
				Page:       page,
				PerPage:    perPage,
				TotalItems: total,
				TotalPages: totalPages,
				HasNext:    page < totalPages,
				HasPrev:    page > 1,
			},
		},
		Meta: &Meta{
			RequestID: c.GetHeader("X-Request-ID"),
			Timestamp: currentTime(),
		},
	})
}

func Error(c *gin.HContext, httpCode int, errCode int, message string) {
	c.JSON(httpCode, gin.H{
		"code":    errCode,
		"error":   errorCodeToMessage(errCode),
		"message": message,
		"request_id": c.GetHeader("X-Request-ID"),
	})
}

func ValidationError(c *gin.HContext, err error) {
	Error(c, http.StatusBadRequest, 40001, err.Error())
}

func UnauthorizedError(c *gin.HContext, message string) {
	Error(c, http.StatusUnauthorized, 70001, message)
}

func NotFoundError(c *gin.HContext, resource string) {
	Error(c, http.StatusNotFound, 40004, resource+" not found")
}

func InternalError(c *gin.HContext, err error) {
	Error(c, http.StatusInternalServerError, 50000, "Internal server error")
	// 生产环境应记录日志但不暴露细节
	if os.Getenv("APP_ENV") == "development" {
		gin.H{"detail": err.Error()}["detail"] = err.Error()
	}
}

func currentTime() string {
	return time.Now().UTC().Format("2006-01-02T15:04:05Z07:00")
}

func errorCodeToMessage(code int) string {
	messages := map[int]string{
		40001: "VALIDATION_ERROR",
		40002: "INVALID_FILE_TYPE",
		40003: "FILE_SIZE_EXCEEDED",
		40004: "RESOURCE_NOT_FOUND",
		40101: "AUTH_TOKEN_EXPIRED",
		40102: "INVALID_TOKEN",
		40103: "INSUFFICIENT_PERMISSIONS",
		40901: "DUPLICATE_RESOURCE",
		42900: "RATE_LIMIT_EXCEEDED",
		50000: "INTERNAL_ERROR",
	}
	if msg, ok := messages[code]; ok {
		return msg
	}
	return "UNKNOWN_ERROR"
}
```

```go
package middleware

import (
	"net/http"
	"strings"
	"time"

	"github.com/gin-gonic/gin"
	"github.com/google/uuid"
	"github.com/smartwardrobe/api-server/pkg/response"

// CORS 跨域中间件
func CORS() gin.HandlerFunc {
	return func(c *gin.Context) {
		c.Writer.Header().Set("Access-Control-Allow-Origin", "*")
		c.Writer.Header().Set("Access-Control-Allow-Credentials", "true")
		c.Writer.Header().Set("Access-Control-Allow-Headers", "Content-Type, Content-Length, Accept-Encoding, X-CSRF-Token, Authorization, X-Request-ID, accept, origin, Cache-Control, X-Requested-With")
		c.Writer.Header().Set("Access-Control-Allow-Methods", "POST, OPTIONS, GET, PUT, DELETE, PATCH")

		if c.Request.Method == "OPTIONS" {
			c.AbortWithStatus(http.StatusNoReturn)
			return
		}

		c.Next()
	}
}

// RequestID 为每个请求生成唯一ID
func RequestID() gin.HandlerFunc {
	return func(c *gin.Context) {
		requestID := c.GetHeader("X-Request-ID")
		if requestID == "" {
			requestID = uuid.New().String()
		}
		c.Set("request_id", requestID)
		c.Header("X-Request-ID", requestID)
		c.Next()
	}
}

// Logger 结构化日志中间件
func Logger() gin.HandlerFunc {
	return func(c *gin.Context) {
		start := time.Now()
		path := c.Request.URL.Path
		query := c.Request.URL.RawQuery

		c.Next()

		latency := time.Since(start())
		status := c.Writer.Status()

		// 只记录非健康检查请求
		if path != "/healthz" {
			log.Printf("[%s] %s %s %s %v",
				status,
				c.Request.Method,
				path+query,
				latency,
			)
		}
	}
}

// Recovery panic恢复
func Recovery() gin.HandlerFunc {
	return func(c *gin.Context) {
		defer func() {
			if err := recover(); err != nil {
				response.InternalError(c, err.(error))
				c.Abort()
			}
		}()
		c.Next()
	}
}
```

#### 10.6 认证中间件 `internal/middleware/auth.go`

```go
package middleware

import (
	"net/http"
	"strings"

	"github.com/gin-gonic/gin"
	"github.com/golang-jwt/jwt/v5"
	"github.com/smartwardrobe/api-server/pkg/response"
)

var jwtSecret = []byte("smartwardrobe-dev-secret-change-in-production") // 应从配置读取

// AuthRequired JWT认证中间件
func AuthRequired() gin.HandlerFunc {
	return func(c *gin.Context) {
		authHeader := c.GetHeader("Authorization")
		if authHeader == "" {
			response.UnauthorizedError(c, "Missing authorization header")
			c.Abort()
			return
		}

		parts := strings.SplitN(authHeader, " ", 2)
		if len(parts) != 2 || parts[0] != "Bearer" {
			response.UnauthorizedError(c, "Invalid authorization format")
			c.Abort()
			return
		}

		tokenString := parts[1]
		token, err := jwt.Parse(tokenString, func(token *jwt.Token) (interface{}, error) {
			return jwtSecret, nil
		})

		if err != nil || !token.Valid {
			response.UnauthorizedError(c, "Invalid or expired token")
			c.Abort()
			return
		}

		claims, ok := token.Claims.(jwt.MapClaims)
		if !ok {
			response.UnauthorizedError(c, "Invalid token claims")
			c.Abort()
			return
		}

		userID, exists := claims["sub"]
		if !exists {
			response.UnauthorizedError(c, "Missing user ID in token")
			c.Abort()
			return
		}

		// 将userID存入context供后续handler使用
		c.Set("user_id", userID.(string))
		c.Next()
	}
}

// GetUserID 从context获取当前用户ID
func GetUserID(c *gin.Context) (string, bool) {
	userID, exists := c.Get("user_id")
	if !exists {
		return "", false
	}
	return userID.(string), true
}
```

```go
type AuthHandler struct {
	db *gorm.DB

func NewAuthHandler(db *gorm.DB) *AuthHandler {
	return &AuthHandler{db: db}
}

type registerRequest struct {
	Email    string `json:"email" binding:"required,email"`
	Password string `json:"password" binding:"required,min=6"`
	Nickname string `json:"nickname" binding:"required,min=2,max=50"`
}

type loginRequest struct {
	Email    string `json:"email" binding:"required,email"`
	Password string `json:"password" binding:"required"`
}

// POST /v1/auth/register
func (h *AuthHandler) Register(c *gin.Context) {
	var req registerRequest
	if err := c.ShouldBindJSON(&req); err != nil {
		response.ValidationError(c, err)
		return
	}

	// 检查邮箱是否已存在
	var existingUser model.User
	if result := h.db.Where("email = ?", req.Email).First(&existingUser); result.RowsAffected > 0 {
		response.Error(c, http.StatusConflict, 40901, "Email already registered")
		return
	}

	// 加密密码
	hashedPassword, err := bcrypt.GenerateFromPassword([]byte(req.Password), bcrypt.DefaultCost)
	if err != nil {
		response.InternalError(c, err)
		return
	}

	// 创建用户
	user := model.User{
		Email:        req.Email,
		PasswordHash: string(hashedPassword),
		Nickname:     req.Nickname,
		IsActive:     true,
	}

	if err := h.db.Create(&user).Error; err != nil {
		response.InternalError(c, err)
		return
	}

	// 生成Token
	tokenData := generateTokenPair(user.ID)

	response.Created(c, gin.H{
		"access_token":  tokenData.AccessToken,
		"refresh_token": tokenData.RefreshToken,
		"expires_in":     86400, // 24小时
		"user": gin.H{
			"id":           user.ID.String(),
			"email":        user.Email,
			"nickname":     user.Nickname,
			"avatar_url":   user.AvatarURL,
			"plan_type":    "free",
			"created_at":   user.CreatedAt.Format(time.RFC3339),
		},
	})
}

// POST /v1/auth/login
func (h *AuthHandler) Login(c *gin.Context) {
	var req loginRequest
	if err := c.ShouldBindJSON(&req); err != nil {
		response.ValidationError(c, err)
		return
	}

	var user model.User
	if err := h.db.Where("email = ? AND is_active = ?", req.Email, true).First(&user).Error; err != nil {
		response.UnauthorizedError(c, "Invalid email or password")
		return
	}

	// 验证密码
	if err := bcrypt.CompareHashAndPassword([]byte(req.Password), []byte(user.PasswordHash)); err != nil {
		response.UnauthorizedError(c, "Invalid email or password")
		return
	}

	// 生成Token
	tokenData := generateTokenPair(user.ID)

	response.Success(c, gin.H{
		"access_token":  tokenData.AccessToken,
		"refresh_token": tokenData.RefreshToken,
		"expires_in":     86400,
		"user": gin.H{
			"id":         user.ID.String(),
			"email":      user.Email,
			"nickname":   user.Nickname,
			"avatar_url": user.AvatarURL,
			"plan_type":  "free",
			"created_at": user.CreatedAt.Format(time.RFC3339),
		},
	})
}

		"avatar_url": user.AvatarURL,
		"plan_type":  "free",
		"created_at": user.CreatedAt.Format(time.RFC3339),
	})
}

// Token生成辅助函数
type TokenPair struct {
	AccessToken  string
	RefreshToken string
}

func generateTokenPair(userID uuid.UUID) TokenPair {
	now := time.Now()

	// Access Token (短期)
	accessToken := jwt.NewWithClaims(jwt.SigningMethod{}, jwt.MapClaims{
		"sub": userID.String(),
		"exp": now.Add(24 * time.Hour).Unix(),
		"iat": now.Unix(),
		"type": "access",
	})
	accessTokenString, _ := accessToken.SignedString(jwtSecret)

	// Refresh Token (长期)
	refreshToken := jwt.NewWithClaims(jwt.SigningMethod{}, jwt.MapClaims{
		"sub": userID.String(),
		"exp": now.Add(7 * 24 * time.Hour).Unix(), // 7天
		"iat": now.Unix(),
		"type": "refresh",
	})
	refreshTokenString, _ := refreshToken.SignedString(jwtSecret)

	return TokenPair{
		AccessToken:  accessTokenString,
		RefreshToken: refreshTokenString,
	}
}
```

```go
// GET /v1/clothes
func (h *ThingHandler) List(c *gin.Context) {
	userID, _ := middleware.GetUserID(c)

	page, _ := strconv.Atoi(c.DefaultQuery("page", "1"))
	perPage, _ := strconv.Atoi(c.DefaultQuery("per_page", "20"))
	category := c.Query("category")
	folderID := c.Query("folder_id")
	search := c.Query("search")
	showFavoritesOnly := c.Query("is_favorite") == "true"

	if page < 1 {
		page = 1
	}
	if perPage < 1 || perPage > 100 {
		perPage = 20
	}

	query := h.db.Model(&model.Clothing{}).Where("user_id = ?", userID)

	if category != "" {
		query = query.Where("category = ?", category)
	}
	if folderID != "" {
		folderUUID, err := uuid.Parse(folderID)
		if err == nil {
			query = query.Where("folder_id = ?", folderUUID)
		}
	}
	if search != "" {
		query = query.Where("name ILIKE ? OR brand ILIKE ? OR custom_tags::text ILIKE ?",
			"%"+search+"%", "%"+search+"%", "%"+search+"%")
	}
	if showFavoritesOnly {
		query = query.Where("is_favorite = ?", true)
	}

	var total int64
	query.Count(&total)

	var clothes []model.Clothing
	offset := (page - 1) * perPage
	query.Order("created_at DESC").Offset(offset).Limit(perPage).Find(&clothes)

	response.Paginated(c, clothes, page, perPage, int(total))
}

// GET /v1/clothes/:id
func (h *ThingHandler) GetByID(c *gin.Context) {

// POST /v1/clothes/upload
func (h *ThingHandler) Upload(c *gin.Context) {
	userID, _ := middleware.GetUserID(c)
	userUUID, _ := uuid.Parse(userID)

	form, err := c.MultipartForm()
	if err != nil {
		response.ValidationError(c, err)
		return
	}

	files := form.File["images"]
	if len(files) == 0 {
		response.ValidationError(c, err)
		return
	}

	// 解析基本字段
	category := form.Value["category"][0]
	primaryColor := form.Value["primary_color"][0]
	name := form.Value["name"][0]
	brand := form.Value["brand"][0]
	priceStr := form.Value["price"][0]
	seasons := form.Value["seasons[]"]

	var price *float64
	if priceStr != "" && priceStr[0] != "" {
		p, err := strconv.ParseFloat64(priceStr[0], 64)
		if err == nil {
			price = &p
		}
	}

	// 处理第一张图片作为主图
	file := files[0]
	filename := uuid.New().String() + filepath.Ext(file.Filename)
	savePath := filepath.Join(h.uploadDir, filename)

	if err := c.SaveUploadedFile(file, savePath); err != nil {
		response.Error(c, http.StatusInternalServerError, 45002, "Failed to save image")
		return
	}

	imageURL := "/uploads/images/" + filename
	imageHash := computeFileHash(savePath) // 简化的hash计算

	// 创建衣物记录
	clothing := model.Clothing{
		UserID:       userUUID,
		Name:         name,
		Category:     category,
		PrimaryColor: primaryColor,
		Brand:        brand,
		Price:        price,
		ImageURL:     imageURL,
		ThumbnailURL: imageURL, // v1.0暂无缩略图生成
		ImageHash:    imageHash,
		Status:       "active",
		Seasons:      seasons,
	}

	if err := h.db.Create(&clothing).Error; err != nil {
		response.InternalError(c, err)
		return
	}

	response.Created(c, gin.H{
		"clothing_ids": []string{clothing.ID.String()},
		"processed_images": []gin.H{
			"clothing_id": clothing.ID.String(),
			"original_url": imageURL,
			"thumbnail_url": imageURL,
		},
		"upload_status": "success",
	})
}
```
