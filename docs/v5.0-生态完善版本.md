## 版本概述
**发布日期**: 2027-Q1
**开发周期**: 14周

**目标**: 构建完整生态系统，实现社交、AR、购物集成与开放平台能力

## 一、版本定位：从工具到生活方式平台

### 演进终态
```
v1.0-v2.0: 数字化工具 → 管理个人衣物
v3.0: AI助手 → 智能决策支持
v4.0: 生活管家 → 全生命周期管理
v5.0: 时尚生态 → 社交+AR+购物+开发者平台
```

### 核心愿景
**SmartWardrobe 不再只是一个App，而是连接人与时尚的智能中枢**

- **对用户**: 一站式时尚生活解决方案（管理-搭配-购买-分享）
- **对品牌/商家**: 触达精准用户的营销渠道
- **对开发者**: 构建时尚科技应用的底层基础设施

# services/community/feed_algorithm.py
class SocialFeedGenerator:
    """
    多信号融合的社区内容推荐
    """
    def __init__(self):
        self.content_based = ContentBasedRecommender(
            embedding_model='fashion_clip',  # 专门的时尚领域CLIP模型
            feature_dim=512
        )
        self.collaborative = SocialCollaborativeFiltering(
            graph_based=True,  # 使用图神经网络建模社交关系
        )
        self.trend_detector = FashionTrendDetector(
            data_sources=['instagram', 'xiaohongshu', 'runway_shows']
        )

    async def generate_feed(self, user_id: str, page_size: int = 20) -> FeedResponse:
        # Signal 1: 关注的人的动态（权重40%）
        following_posts = await self._get_following_posts(user_id, limit=int(page_size * 0.4))

        # Signal 2: 基于风格偏好的相似内容（权重30%）
        style_matched = await self.content_based.recommend_by_user_style(user_id, limit=int(page_size * 0.3))

        # Signal 3: 热门/趋势内容（权重20%）
        trending = await self.trend_detector.get_trending_posts(geo=user_location, limit=int(page_size * 0.2))

        # Signal 4: 新人曝光池（权重10%，扶持优质新创作者）
        new_creator_spotlight = await self._get_quality_new_posts(limit=int(page_size * 0.1))

        # 融合与去重
        merged_feed = self.merge_and_deduplicate([
            following_posts, style_matched, trending, new_creator_spotlight
        ])

        # 重排序：多样性保证 + 新鲜度衰减 + 业务规则插入
        reranked = self.rerank_with_diversity(merged_feed)

        return FeedResponse(posts=reranked, feed_id=generate_session_id())

#### AR试衣核心流程

```typescript
// services/ar/real_time_tryon.ts
class ARRealTimeTryOn {
  private poseEstimator: MediaPipePose;
  private garmentRenderer: WebGLGarmentOverlay;
  private bodySegmenter: BodySegmentationNet;

  async initialize(cameraStream: MediaStream): Promise<void> {
    // 初始化MediaPipe人体姿态估计（33个关键点）
    this.poseEstimator = new Pose({
      locateFile: (file) => `https://cdn.jsdelivr.net/npm/@mediapipe/pose/${file}`
    });
    await this.poseEstimator.setOptions({
      modelComplexity: 1,
      smoothLandmarks: true,
      enableSegmentation: true,  // 同时输出人体分割mask
      minDetectionConfidence: 0.7,
      minTrackingConfidence: 0.7
    });

    // 初始化WebGL渲染器（用于叠加衣物纹理）
    this.garmentRenderer = new WebGLGarmentRenderer(canvasElement);
  }

  async startTryOnSession(
    selectedGarments: GarmentSelection[],  // 用户选择的衣物组合
    cameraElement: HTMLVideoElement
  ): Promise<void> {
    const processFrame = async () => {
      // Step 1: 实时检测人体关键点（<30ms）
      const results = await this.poseEstimator.send({ image: cameraElement });

      if (results.poseLandmarks && results.segmentationMask) {
        // Step 2: 提取关键部位坐标
        const upperBodyKeypoints = this.extractUpperBodyPoints(results.poseLandmarks);
        const lowerBodyKeypoints = this.extractLowerBodyPoints(results.poseLandmarks);

        // Step 3: 根据姿态形变衣物纹理（仿射变换+网格变形）
        const warpedUpperTexture = this.warpGarmentTexture(
          selectedGarments.upper?.texture,
          upperBodyKeypoints
        );
        const warpedLowerTexture = this.warpGarmentTexture(
          selectedGarments.lower?.texture,
          lowerBodyKeypoints
        );

        // Step 4: WebGL渲染合成画面（<16ms, 达到60fps）
        this.garmentRenderer.renderFrame({
          background: cameraElement,
          segmentationMask: results.segmentationMask,
          layers: [
            { texture: warpedUpperTexture, zIndex: 2 },
            { texture: warpedLowerTexture, zIndex: 1 }
          ]
        });
      }

      requestAnimationFrame(processFrame);  // 循环处理下一帧
    };

    processFrame();
  }
}
```

// components/ARTryOnCamera.tsx
function ARTryOnCamera({ garments }: Props) {
  const videoRef = useRef<HTMLVideoElement>(null);
  const canvasRef = useRef<HTMLCanvasElement>(null);
  const [isInitialized, setIsInitialized] = useState(false);
  const [fps, setFps] = useState(0);

  useEffect(() => {
    const initAR = async () => {
      try {
        const arEngine = new ARRealTimeTryOn();

        // 请求摄像头权限
        const stream = await navigator.mediaDevices.getUserMedia({
          video: { facingMode: 'user', width: 1280, height: 720 }
        });
        videoRef.current.srcObject = stream;
        await videoRef.current.play();

        // 初始化AR引擎
        await arEngine.initialize(stream);

        // 启动实时试衣
        await arEngine.startTryOnSession(garments, videoRef.current);

        setIsInitialized(true);

        // 性能监控
        setInterval(() => {
          setFps(arEngine.getCurrentFPS());
        }, 1000);
      } catch (error) {
        showError("AR初始化失败，请检查设备兼容性");
      }
    };

    initAR();

    return () => {
      // 清理资源
      arEngine?.destroy();
      videoRef.current?.srcObject.getTracks().forEach(t => t.stop());
    };
  }, [garments]);

  return (
    <div className="ar-tryon-container">
      <video ref={videoRef} autoPlay playsInline muted className="hidden" />
      <canvas ref={canvasRef} className="ar-canvas" />

      {/* UI覆盖层 */}

#### 购物助手核心逻辑

```python
# services/shopping/assistant.py
class ShoppingAssistant:
    def __init__(self):
        self.product_recognizer = ProductRecognitionModel()
        self.duplicate_checker = WardrobeDuplicateDetector()
        self.stylist_engine = OutfitStylistEngine()
        self.price_monitor = PriceTracker()

    async def analyze_purchase_intention(
        self,
        user_id: str,
        product_input: ProductInput  # URL or Image
    ) -> PurchaseAnalysis:

        # Step 1: 商品识别与属性提取
        product_info = await self.product_recognizer.recognize(product_input)
        # 返回: {brand, category, color, price, material, style_tags, image_urls}

        # Step 2: 衣橱查重（核心价值点！）
        duplicate_check = await self.duplicate_checker.check(
            user_id=user_id,
            target_product=product_info,
            similarity_threshold=0.85  # 85%以上视为高度疑似重复
        )

        analysis = PurchaseAnalysis(product=product_info)

        if duplicate_check.is_duplicate:
            analysis.verdict = 'DUPLICATE_FOUND'
            analysis.existing_items = duplicate_check.matched_items
            analysis.message = f"您衣橱中已有{len(duplicate_check.matched_items)}件相似单品"
            analysis.suggestion = "建议考虑其他款式以增加衣橱多样性"

        elif duplicate_check.is_similar:
            analysis.verdict = 'SIMILAR_EXISTS'
            analysis.similar_items = duplicate_check.similar_items
            analysis.message = "您有类似风格的单品，但这件有独特之处"
            analysis.unique_selling_points = self._analyze_uniqueness(
                product_info, duplicate_check.similar_items
            )

        else:
            analysis.verdict = 'NEW_ITEM'
            analysis.message = "✨ 这件与您现有的衣物形成良好互补"

            # Step 3: 如果是新物品，生成搭配预览
            outfit_preview = await self.stylist_engine.generate_preview_with_new_item(
                user_id=user_id,
                new_item=product_info,
                existing_wardrobe=await self.get_user_wardrobe(user_id)
            )
            analysis.outfit_previews = outfit_preview  # 展示3套搭配方案

        # Step 4: 价格评估
        price_eval = await self.price_monitor.evaluate_price(
            product=product_info,
            category_market_data=await self.get_market_prices(product_info.category)
        )
        analysis.price_assessment = price_eval

        # Step 5: 应用一进一出策略（如果启用）
        if await self.is_one_in_one_out_enabled(user_id):
            analysis.swap_suggestion = await self.suggest_swap_candidates(
                user_id,
                new_item_category=product_info.category
            )

        return analysis
```

    USING (
        EXISTS (
            SELECT 1 FROM tenant_members tm
            JOIN tenants t ON tm.tenant_id = t.id

              AND t.id = current_setting('app.current_tenant_id')::uuid

## 八、已知限制与技术债务

1. **社区内容审核压力**: UGC内容需要大量审核资源（初期人工+后期AI辅助）
2. **AR设备碎片化**: Android机型众多，低端机体验差（需明确最低配置要求）
3. **第三方平台依赖**: 购物助手依赖电商平台API稳定性（反爬风险）
4. **多租户数据隔离复杂性**: RLS策略可能影响查询性能（需充分压测）
5. **法律合规成本**: GDPR/PIPL等法规在不同国家合规成本高昂
6. **开放平台滥用风险**: 需建立完善的API滥用检测与封禁机制
