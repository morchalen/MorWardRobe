### 1.2 数据库集群拓扑（生产环境）

```
                    ┌─────────────────┐
                    │   Write Proxy   │  (PgBouncer / Patroni)
                    │   (Primary RW)  │
                    └────────┬────────┘
                             │
                    ┌────────▼────────┐
                    │   Primary Node  │  (主库，接受写操作)
                    │   PostgreSQL 15 │
                    └────────┬────────┘
                             │ 同步复制
              ┌──────────────┼──────────────┐
              ▼              ▼              ▼
     ┌────────────┐  ┌────────────┐  ┌────────────┐
     │ Replica-1  │  │ Replica-2  │  │ Replica-3  │
     │ (Read Only)│  │ (Read Only)│  │ (Read Only)│
     └────────────┘  └────────────┘  └────────────┘
          读负载            分析查询           社区Feed查询
```

### 1.3 技术选型理由

| 组件 | 选择 | 理由 |
|------|------|------|
| **主数据库** | PostgreSQL 15+ | JSONB支持好、GIS扩展(地理位置)、强大的全文搜索、成熟稳定 |
| **缓存层** | Redis 7 | 会话管理、热点数据缓存、消息队列 |
| **对象存储** | MinIO (私有) / S3 (公有) | 图片/模型文件存储，兼容S3协议 |
| **搜索引擎** | Meilisearch | 轻量级全文检索，比Elasticsearch易维护 |
| **向量数据库** | pgvector (初期) → Milvus (规模化后) | 视觉相似度检索，pgvector适合中小规模 |

```sql
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID NOT NULL REFERENCES sw_core.users(id) ON DELETE CASCADE,

    name VARCHAR(100) NOT NULL,
    description TEXT,
    icon VARCHAR(50),                   -- emoji或图标名
    color VARCHAR(7),                   -- HEX

    parent_folder_id UUID REFERENCES sw_core.folders(id) ON DELETE CASCADE,
    sort_order INTEGER DEFAULT 0,

    is_system BOOLEAN DEFAULT false,
    system_type VARCHAR(30)
        CHECK (system_type IN ('all','favorites','recently_added',
                               'season_spring','season_summer','season_autumn','season_winter',
                               'to_wash','idle')),

    clothing_count INTEGER NOT NULL DEFAULT 0,

    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),

    CONSTRAINT uq_user_folder_name UNIQUE (user_id, name, parent_folder_id)
);

CREATE INDEX idx_folders_user_parent ON sw_core.folders(user_id, parent_folder_id);
CREATE INDEX idx_folders_system ON sw_core.folders(user_id, system_type) WHERE is_system = true;

-- 递归CTE辅助：获取文件夹完整路径
CREATE OR REPLACE FUNCTION sw_core.get_folder_path(folder_id UUID)
RETURNS TABLE (path_element UUID, level INT) AS $$
BEGIN
    RETURN QUERY
    WITH RECURSIVE folder_tree AS (
        SELECT id, parent_folder_id, 0 as level
        FROM sw_core.folders WHERE id = folder_id
        UNION ALL
        SELECT f.id, f.parent_folder_id, ft.level + 1
        FROM sw_core.folders f
        JOIN folder_tree ft ON f.id = ft.parent_folder_id
    )
    SELECT id, level FROM folder_tree ORDER BY level DESC;
END;
$$ LANGUAGE plpgsql STABLE;
```

## 六、存储过程与函数

### 6.1 衣物状态转换（带业务逻辑校验）

```sql
CREATE OR REPLACE FUNCTION sw_core.fn_transition_clothing_status(
    p_clothing_id UUID,
    p_target_status VARCHAR(20),
    p_reason TEXT DEFAULT NULL,
    p_triggered_by VARCHAR(30) DEFAULT 'user_manual'
)
RETURNS JSONB AS $$
DECLARE
    v_current_status VARCHAR(20);
    v_user_id UUID;
    v_is_valid BOOLEAN := false;
BEGIN
    -- 1. 获取当前状态和用户
    SELECT status, user_id INTO v_current_status, v_user_id
    FROM sw_core.clothes
    WHERE id = p_clothing_id FOR UPDATE;  -- 行锁防止并发冲突

    IF NOT FOUND THEN
        RAISE EXCEPTION 'Clothing not found: %', p_clothing_id;
    END IF;

    -- 2. 校验状态转换合法性（基于预定义的状态机规则）
    SELECT EXISTS(
        SELECT 1 FROM (VALUES
            ('active', ARRAY['pending_wash','seasonal_storage','idle','discarded']),
            ('pending_wash', ARRAY['drying','dry_cleaning','active']),
            ('drying', ARRAY['active']),
            ('dry_cleaning', ARRAY['active']),
            ('seasonal_storage', ARRAY['active']),
            ('idle', ARRAY['active','donation_ready','resale_ready','discarded']),
            ('donation_ready', ARRAY[]),  -- 终态
            ('resale_ready', ARRAY[]),    -- 终态
            ('discarded', ARRAY[])        -- 终态
        ) AS valid(from_state, allowed_targets)
        WHERE valid.from_state = v_current_status
          AND p_target_status = ANY(valid.allowed_targets)
    ) INTO v_is_valid;

    IF NOT v_is_valid THEN
        RAISE EXCEPTION 'Invalid status transition from % to %', v_current_status, p_target_status;
    END IF;

    -- 3. 执行状态更新
    UPDATE sw_core.clothes
    SET status = p_target_status,
        updated_at = NOW()
    WHERE id = p_clothing_id;

    -- 4. 记录审计日志
    INSERT INTO sw_core.clothing_status_logs (clothing_id, user_id, from_status, to_status, reason, triggered_by)
    VALUES (p_clothing_id, v_user_id, v_current_status, p_target_status, p_reason, p_triggered_by);

    -- 5. 特殊状态的附加处理
    IF p_target_status = 'idle' THEN
        PERFORM sw_core.fn_trigger_idle_analysis(p_clothing_id);
    ELSIF p_target_status = 'seasonal_storage' THEN
        PERFORM sw_core.fn_update_folder_counts(v_user_id);
    END IF;

    -- 返回成功结果
    RETURN jsonb_build_object(
        'success', true,
        'clothing_id', p_clothing_id,
        'from_status', v_current_status,
        'to_status', p_target_status,
        'timestamp', NOW()
    );
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;
```

### 6.2 衣橱容量检查（一进一出策略）

```sql
CREATE OR REPLACE FUNCTION sw_core.fn_check_wardrobe_capacity(
    p_user_id UUID,
    p_new_items_count INTEGER DEFAULT 1
) RETURNS TABLE (
    allowed BOOLEAN,
    current_count INTEGER,
    max_capacity INTEGER,
    suggested_removals JSONB
) AS $$
DECLARE
    v_capacity_limit INTEGER;
    v_current_active INTEGER;
BEGIN
    -- 获取用户的衣橱上限（根据套餐不同）
    SELECT CASE plan_type
        WHEN 'free' THEN 200
        WHEN 'pro_individual' THEN 2000
        WHEN 'pro_family' THEN 5000
        ELSE 10000  -- business/enterprise
    END INTO v_capacity_limit
    FROM sw_core.users WHERE id = p_user_id;

    -- 当前活跃数量
    SELECT COUNT(*) INTO v_current_active
    FROM sw_core.clothes
    WHERE user_id = p_user_id AND status = 'active';

    -- 判断是否超限
    allowed := (v_current_active + p_new_items_count <= v_capacity_limit);

    -- 如果超限，建议移出的物品（基于闲置评分）
    IF NOT allowed THEN
        suggested_removals := (
            SELECT jsonb_agg(jsonb_build_object(
                'id', id,
                'name', name,
                'image_url', thumbnail_url,
                'idle_days', COALESCE(idle_days_count, EXTRACT(DAY FROM (NOW() - last_wear_date))::INTEGER),
                'reason', CASE
                    WHEN idle_days_count > 90 THEN '超过90天未穿'
                    WHEN idle_days_count > 60 THEN '超过60天未穿'
                    ELSE '低频使用单品'
                END
            ))
            FROM sw_core.clothes
            WHERE user_id = p_user_id
              AND status = 'active'
            ORDER BY
                CASE WHEN last_worn_date IS NULL THEN 0 ELSE 1 END,
                last_worn_date ASC NULLS FIRST
            LIMIT p_new_items_count
        );
    END IF;

    current_count := v_current_active;
    max_capacity := v_capacity_limit;

    RETURN;
END;
$$ LANGUAGE plpgsql STABLE;
```

```sql
CREATE OR REPLACE FUNCTION sw_core.fn_sync_folder_clothing_count()
RETURNS TRIGGER AS $$
BEGIN
    IF TG_OP = 'INSERT' AND NEW.folder_id IS NOT NULL THEN
        UPDATE sw_core.folders SET clothing_count = clothing_count + 1
        WHERE id = NEW.folder_id;
    ELSIF TG_OP = 'DELETE' AND OLD.folder_id IS NOT NULL THEN
        UPDATE sw_core.folders SET clothing_count = clothing_count - 1
        WHERE id = OLD.folder_id;
    ELSIF TG_OP = 'UPDATE' THEN
        IF OLD.folder_id IS DISTINCT FROM NEW.folder_id THEN
            IF OLD.folder_id IS NOT NULL THEN
                UPDATE sw_core.folders SET clothing_count = clothing_count - 1 WHERE id = OLD.folder_id;
            END IF;
            IF NEW.folder_id IS NOT NULL THEN
                UPDATE sw_core.folders SET clothing_count = clothing_count + 1 WHERE id = NEW.folder_id;
            END IF;
        END IF;
    END IF;

    IF TG_OP IN ('INSERT', 'UPDATE') THEN RETURN NEW;
    ELSE RETURN OLD; END IF;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_clothes_folder_count_sync
    AFTER INSERT OR UPDATE OR DELETE ON sw_core.clothes
```

```sql
END;
$$ LANGUAGE plpgsql;
```

## 十、性能监控SQL

### 10.1 健康检查Dashboard查询

```sql
-- 1. 数据库连接数
SELECT state, count(*) FROM pg_stat_activity GROUP BY state;

-- 2. 慢查询Top 10
SELECT query, calls, total_exec_time, mean_exec_time, rows
FROM pg_stat_statements
ORDER BY mean_exec_time DESC LIMIT 10;

-- 3. 表膨胀率（需要VACUUM的表）
SELECT
    schemaname || '.' || relname AS table_name,
    n_dead_tup AS dead_rows,
    n_live_tup AS live_rows,
    round(n_dead_tup::numeric /
        CASE WHEN n_live_tup > 0 THEN n_live_tup ELSE 1 END * 100, 2) AS bloat_ratio,
    last_vacuum,
    last_autovacuum
FROM pg_stat_user_tables
WHERE n_dead_tup > 1000
ORDER BY bloat_ratio DESC
LIMIT 20;

-- 4. 索引利用率
SELECT
    schemaname || '.' || relname AS table_name,
    indexrelname AS index_name,
    idx_scan,
    idx_tup_read,
    idx_tup_fetch,
    pg_size_pretty(pg_relation_size(indexrelid)) AS index_size
FROM pg_stat_user_indexes
WHERE idx_scan < 50
ORDER BY pg_relation_size(indexrelid) DESC
LIMIT 20;

-- 5. 缓存命中率（应>99%）
SELECT
    round((sum(blks_hit) / (sum(blks_hit) + sum(blks_read))) * 100, 2) AS cache_hit_ratio
FROM pg_stat_database
WHERE datname = 'smartwardrobe';

-- 6. 锁等待检测
SELECT
    blocked_locks.pid AS blocked_pid,
    blocked_activity.usename AS blocked_user,
    blocking_locks.pid AS blocking_pid,
    blocking_activity.usename AS blocking_user,
    blocked_activity.query AS blocked_statement,
    blocking_activity.query AS current_statement_in_blocking_process,
    blocked_activity.application_name AS blocked_application
FROM pg_catalog.pg_locks blocked_locks
JOIN pg_catalog.stat_activity blocked_activity ON blocked_activity.pid = blocked_locks.pid
JOIN pg_catalog.pg_locks blocking_locks
    ON blocking_locks.locktype = blocked_locks.locktype
    AND blocking_locks.DATABASE IS NOT DISTINCT FROM blocked_locks.DATABASE
    AND blocking_locks.relation IS NOT DISTINCT FROM blocked_locks.relation
    AND blocking_locks.page IS NOT DISTINCT FROM blocked_locks.page
    AND blocking_locks.tuple IS NOT DISTINCT FROM blocked_locks.tuple
    AND blocking_locks.virtualxid IS NOT DISTINCT FROM blocked_locks.virtualxid
    AND blocking_locks.transactionid IS NOT DISTINCT FROM blocked_locks.transactionid
    AND blocking_locks.classid IS NOT DISTINCT FROM blocked_locks.classid
    AND blocking_locks.objid IS NOT DISTINCT FROM blocked_locks.objid
    AND blocking_locks.objsubid IS NOT DISTINCT FROM blocked_locks.objsubid
    AND blocking_locks.pid != blocked_locks.pid
JOIN pg_catalog.stat_activity blocking_activity ON blocking_activity.pid = blocking_locks.pid
WHERE NOT blocked_locks.GRANTED;
```

### 10.2 容量规划查询

```sql
-- 各表数据量及增长趋势
```
